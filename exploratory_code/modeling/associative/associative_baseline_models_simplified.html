3) AIS/Environment Missing-Value Imputation Rule

build_week_fill_map <- function(day_slot_tbl, group_col, date_col = "trip_date", value_col = "slot_day_median", window_days = 3) {
  if (nrow(day_slot_tbl) == 0) return(tibble())
  split_tbl <- split(day_slot_tbl, day_slot_tbl[[group_col]], drop = TRUE)
  out <- lapply(split_tbl, function(gdf) {
    gdf <- gdf[order(gdf[[date_col]]), , drop = FALSE]
    d <- gdf[[date_col]]
    v <- gdf[[value_col]]
    week_fill <- vapply(seq_along(v), function(i) {
      idx <- abs(as.integer(d - d[i])) <= window_days
      val <- suppressWarnings(median(v[idx], na.rm = TRUE))
      ifelse(is.finite(val), val, NA_real_)
    }, numeric(1))
    gdf$week_fill <- week_fill
    gdf[, c(group_col, date_col, "week_fill"), drop = FALSE]
  })
  bind_rows(out)
}

impute_by_local_slot_week <- function(data, var_name, window_days = 3) {
  x <- data
  miss_before <- sum(is.na(x[[var_name]]))
  if (miss_before == 0) {
    return(list(data = x, summary = tibble(variable = var_name, n_missing_before = 0L, n_filled_primary = 0L, n_filled_fallback = 0L, n_missing_after = 0L)))
  }

  # Primary: same local hour within centered 7-day window (±3 days by default).
  primary_daily <- x %>%
    filter(!is.na(.data[[var_name]]), !is.na(trip_date), !is.na(prev_stop_sched_hour)) %>%
    group_by(prev_stop_sched_hour, trip_date) %>%
    summarise(slot_day_median = median(.data[[var_name]], na.rm = TRUE), .groups = "drop")
  primary_fill <- build_week_fill_map(primary_daily, group_col = "prev_stop_sched_hour", value_col = "slot_day_median", window_days = window_days) %>%
    rename(fill_primary = week_fill)

  # Fallback: same local hour-group within centered 7-day window.
  fallback_daily <- x %>%
    filter(!is.na(.data[[var_name]]), !is.na(trip_date), !is.na(prev_stop_sched_hour_group)) %>%
    group_by(prev_stop_sched_hour_group, trip_date) %>%
    summarise(slot_day_median = median(.data[[var_name]], na.rm = TRUE), .groups = "drop")
  fallback_fill <- build_week_fill_map(fallback_daily, group_col = "prev_stop_sched_hour_group", value_col = "slot_day_median", window_days = window_days) %>%
    rename(fill_fallback = week_fill)

  x <- x %>%
    left_join(primary_fill, by = c("prev_stop_sched_hour", "trip_date")) %>%
    left_join(fallback_fill, by = c("prev_stop_sched_hour_group", "trip_date"))

  was_missing <- is.na(x[[var_name]])
  fill_primary_idx <- was_missing & !is.na(x$fill_primary)
  fill_fallback_idx <- was_missing & is.na(x$fill_primary) & !is.na(x$fill_fallback)

  x[[var_name]][fill_primary_idx] <- x$fill_primary[fill_primary_idx]
  x[[var_name]][fill_fallback_idx] <- x$fill_fallback[fill_fallback_idx]
  x <- x %>% select(-fill_primary, -fill_fallback)

  list(
    data = x,
    summary = tibble(
      variable = var_name,
      n_missing_before = miss_before,
      n_filled_primary = sum(fill_primary_idx, na.rm = TRUE),
      n_filled_fallback = sum(fill_fallback_idx, na.rm = TRUE),
      n_missing_after = sum(is.na(x[[var_name]]))
    )
  )
}

4) Helper Functions

prepare_model_df <- function(data, outcome, predictors) {
  data %>% select(all_of(c(outcome, predictors))) %>% drop_na()
}

choose_logistic_threshold <- function(y, p, manual_threshold = NA_real_) {
  if (!is.na(manual_threshold)) {
    return(list(threshold = manual_threshold, mode = "manual"))
  }
  thresholds <- seq(0.01, 0.99, by = 0.01)
  j_tbl <- lapply(thresholds, function(t) {
    pred <- ifelse(p >= t, 1L, 0L)
    tp <- sum(pred == 1L & y == 1L); tn <- sum(pred == 0L & y == 0L)
    fp <- sum(pred == 1L & y == 0L); fn <- sum(pred == 0L & y == 1L)
    sens <- ifelse((tp + fn) > 0, tp / (tp + fn), NA_real_)
    spec <- ifelse((tn + fp) > 0, tn / (tn + fp), NA_real_)
    tibble(threshold = t, sensitivity = sens, specificity = spec, youden_j = sens + spec - 1)
  }) %>% bind_rows() %>% filter(!is.na(youden_j))
  best <- j_tbl %>% arrange(desc(youden_j), abs(threshold - 0.5)) %>% slice(1)
  list(threshold = best$threshold[[1]], mode = "youden_j_auto")
}

fit_linear <- function(data, model_name, outcome, predictors) {
  d <- prepare_model_df(data, outcome, predictors)
  fml <- as.formula(paste(outcome, "~", paste(predictors, collapse = " + ")))
  fit <- lm(fml, data = d)
  pred <- predict(fit, newdata = d)
  rmse <- sqrt(mean((d[[outcome]] - pred)^2))
  g <- glance(fit)
  list(
    metrics = tibble(
      model_name = model_name, model_type = "linear", prior_delay_included = ifelse(grepl("with_prior", model_name), "yes", "no"),
      n = nrow(d), aic = g$AIC, bic = g$BIC, rsq = g$r.squared, adj_rsq = g$adj.r.squared, rmse = rmse,
      pseudo_rsq = NA_real_, accuracy = NA_real_, sensitivity = NA_real_, specificity = NA_real_, auc = NA_real_
    ),
    coefs = tidy(fit) %>% mutate(model_name = model_name, model_type = "linear"),
    formula_text = paste(deparse(fml), collapse = "")
  )
}

fit_logistic <- function(data, model_name, outcome_binary, predictors, manual_threshold = NA_real_) {
  d <- prepare_model_df(data, outcome_binary, predictors)
  fml <- as.formula(paste(outcome_binary, "~", paste(predictors, collapse = " + ")))
  fit <- glm(fml, data = d, family = binomial())
  p <- as.numeric(predict(fit, newdata = d, type = "response"))
  y <- as.integer(d[[outcome_binary]])

  thr <- choose_logistic_threshold(y, p, manual_threshold = manual_threshold)
  pred <- ifelse(p >= thr$threshold, 1L, 0L)

  tp <- sum(pred == 1L & y == 1L); tn <- sum(pred == 0L & y == 0L)
  fp <- sum(pred == 1L & y == 0L); fn <- sum(pred == 0L & y == 1L)
  sens <- ifelse((tp + fn) > 0, tp / (tp + fn), NA_real_)
  spec <- ifelse((tn + fp) > 0, tn / (tn + fp), NA_real_)
  acc <- (tp + tn) / nrow(d)
  null_fit <- glm(as.formula(paste(outcome_binary, "~ 1")), data = d, family = binomial())

  n_pos <- sum(y == 1L); n_neg <- sum(y == 0L)
  auc_rank <- if (n_pos == 0 || n_neg == 0) NA_real_ else {
    r <- rank(p, ties.method = "average")
    (sum(r[y == 1L]) - n_pos * (n_pos + 1) / 2) / (n_pos * n_neg)
  }

  g <- glance(fit)
  list(
    metrics = tibble(
      model_name = model_name, model_type = "logistic", prior_delay_included = ifelse(grepl("with_prior", model_name), "yes", "no"),
      n = nrow(d), aic = g$AIC, bic = g$BIC, rsq = NA_real_, adj_rsq = NA_real_, rmse = NA_real_,
      pseudo_rsq = 1 - as.numeric(logLik(fit) / logLik(null_fit)),
      accuracy = acc, sensitivity = sens, specificity = spec, auc = auc_rank
    ),
    coefs = tidy(fit) %>% mutate(model_name = model_name, model_type = "logistic", odds_ratio = exp(estimate)),
    confusion = tibble(model_name = model_name, threshold = thr$threshold, threshold_mode = thr$mode, tp = tp, tn = tn, fp = fp, fn = fn),
    threshold = tibble(model_name = model_name, threshold = thr$threshold, threshold_mode = thr$mode),
    formula_text = paste(deparse(fml), collapse = "")
  )
}
