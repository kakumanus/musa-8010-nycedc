Definition of Delay

Arrival Time Variance (minutes)
Actual Arrival − Scheduled Arrival

Thresholds - Delayed: > 5 minutes
- Excessively delayed: > 10 minutes

Rationale - Based on NYC Ferry operational definitions
- Chosen for technical feasibility (stable classification, less noise)

make_logical_match <- function(df) {
  match_cols <- names(df)[str_detect(names(df), "match")]
  
  df %>%
    mutate(across(all_of(match_cols), ~ case_when(
      .x %in% c(TRUE, "TRUE", "True", "true", "T", "1", 1) ~ TRUE,
      .x %in% c(FALSE, "FALSE", "False", "false", "F", "0", 0) ~ FALSE,
      TRUE ~ NA
    )))
}

parse_dt_nycf <- function(x, tz_local) {
  x_chr <- as.character(x)
  x_chr <- str_trim(na_if(x_chr, ""))

  has_ampm <- str_detect(x_chr, regex("\\b(am|pm)\\b", ignore_case = TRUE))
  has_ampm[is.na(has_ampm)] <- FALSE

  out <- as.POSIXct(rep(NA, length(x_chr)), tz = tz_local)

  idx_2024 <- !has_ampm & !is.na(x_chr)
  idx_2025 <-  has_ampm & !is.na(x_chr)

  out[idx_2024] <- mdy_hm(x_chr[idx_2024], tz = tz_local, quiet = TRUE)
  out[idx_2025] <- parse_date_time(x_chr[idx_2025], orders = "mdY IMSp", tz = tz_local, quiet = TRUE)

  out
}
